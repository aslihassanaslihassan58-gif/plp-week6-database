import os, shutil, datetime, time
print("=== STEP 1: Enabling WAL Archiving ===")
os.makedirs("archive", exist_ok=True)
os.makedirs("backup", exist_ok=True)
print("Archive folder created")
before_disaster = datetime.datetime.now()
print(f"Time BEFORE disaster: {before_disaster}")
time.sleep(1)
with open("backup/database.txt", "w") as f:
    f.write("Important data: Student records")
print("Data created")
shutil.copy("backup/database.txt", "archive/database_backup.txt")
print("Backup taken")
time.sleep(1)
os.remove("backup/database.txt")
print("DISASTER! Database deleted!")
print("\n=== STEP 3: PITR Recovery ===")
shutil.copy("archive/database_backup.txt", "backup/database_restored.txt")
print("Recovery successful! Data restored")
print("\n=== LAB COMPLETE ===")
